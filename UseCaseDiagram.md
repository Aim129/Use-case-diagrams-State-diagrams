```mermaid
%% Диаграмма вариантов использования системы управления сетью библиотек

usecaseDiagram
    actor Читатель
    actor Библиотекарь
    actor Администратор

    %% Иерархия ролей
    Библиотекарь --|> Читатель
    Администратор --|> Библиотекарь

    %% Варианты использования Читателя
    usecase UC1 as "Регистрация в системе"
    usecase UC2 as "Просмотр доступных книг"
    usecase UC3 as "Поиск книг"
    usecase UC4 as "Бронирование книги"
    usecase UC5 as "Отмена бронирования"
    usecase UC6 as "Просмотр истории бронирований"

    %% Варианты использования Библиотекаря
    usecase UC7 as "Добавление книг"
    usecase UC8 as "Удаление книг"
    usecase UC9 as "Учет выданных/возвращенных книг"
    usecase UC10 as "Просмотр активных бронирований"
    usecase UC11 as "Управление каталогом"

    %% Варианты использования Администратора
    usecase UC12 as "Управление филиалами"
    usecase UC13 as "Управление учетными записями"
    usecase UC14 as "Просмотр аналитики"

    %% Акторы и их use cases
    Читатель --> UC1
    Читатель --> UC2
    Читатель --> UC3
    Читатель --> UC4
    Читатель --> UC5
    Читатель --> UC6

    Библиотекарь --> UC7
    Библиотекарь --> UC8
    Библиотекарь --> UC9
    Библиотекарь --> UC10
    Библиотекарь --> UC11

    Администратор --> UC12
    Администратор --> UC13
    Администратор --> UC14

    %% Includes
    UC3 <..> UC2 : <<include>>  %% Поиск всегда включает просмотр
    
    UC7 <..> UC11 : <<include>>  %% Добавление книг — часть управления каталогом
    UC8 <..> UC11 : <<include>>  %% Удаление книг — часть управления каталогом

    %% Extends
    UC4 <.. UC2 : <<extend>>  %% Бронирование возможно после просмотра
    UC5 <.. UC4 : <<extend>>  %% Отмена возможна только после бронирования

    UC10 <.. UC4 : <<extend>>  %% Активные бронирования связаны с бронированием

```
